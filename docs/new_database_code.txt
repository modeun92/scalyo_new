-- ============================================================
-- ENUMS
-- ============================================================

CREATE TYPE person_title AS ENUM (
    'MR',
    'MS',
    'MRS',
    'DR',
    'MX'
);

CREATE TYPE issue_status AS ENUM (
    'OPEN',
    'IN_PROGRESS',
    'RESOLVED',
    'CLOSED'
);

CREATE TYPE authority AS ENUM (
    'VIEW',
    'CREATE',
    'UPDATE',
    'DELETE'
);

CREATE TYPE subscription_type AS ENUM (
    'FREE',
    'BASIC',
    'PRO',
    'ENTERPRISE'
);

CREATE TYPE job_status AS ENUM (
    'ACTIVE',
    'INACTIVE',
    'ON_LEAVE',
    'ENDED'
);

CREATE TYPE client_status AS ENUM (
    'ACTIVE',
    'INACTIVE',
    'CHURNED'
);

-- ============================================================
-- LOOKUP TABLES
-- ============================================================

CREATE TABLE country (
    code TEXT PRIMARY KEY,
    name TEXT NOT NULL
);

CREATE TABLE currency (
    code TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    symbol TEXT
);

CREATE TABLE language_region (
    code TEXT PRIMARY KEY,
    name TEXT NOT NULL
);

-- ============================================================
-- COMPANY
--
-- Base class of Organization and ClientGroup.
-- ============================================================

CREATE TABLE company (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    name TEXT NOT NULL,

    country_code TEXT NOT NULL
        REFERENCES country(code),

    currency_code TEXT NOT NULL
        REFERENCES currency(code),

    photo_path TEXT
);

CREATE TABLE company_link (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    company_id BIGINT NOT NULL
        REFERENCES company(id)
        ON DELETE CASCADE,

    url TEXT NOT NULL,
    sort_order INTEGER NOT NULL DEFAULT 0,

    UNIQUE (company_id, url)
);

-- ============================================================
-- ORGANIZATION : COMPANY
-- ============================================================

CREATE TABLE organization (
    company_id BIGINT PRIMARY KEY
        REFERENCES company(id)
        ON DELETE CASCADE
);

-- ============================================================
-- CLIENT GROUP : COMPANY
--
-- ClientGroup itself belongs to an Organization.
-- This does NOT mean Issue / Profit / Churn belongs to
-- ClientGroup.
-- ============================================================

CREATE TABLE client_group (
    company_id BIGINT PRIMARY KEY
        REFERENCES company(id)
        ON DELETE CASCADE,

    status client_status NOT NULL
);

CREATE TABLE organization_client_group (
    organization_id BIGINT NOT NULL
        REFERENCES organization(company_id)
        ON DELETE CASCADE,

    client_group_id BIGINT NOT NULL
        REFERENCES client_group(company_id)
        ON DELETE CASCADE,

    PRIMARY KEY (
        organization_id,
        client_group_id
    )
);

CREATE INDEX idx_client_group_organization
    ON client_group(organization_id);

-- ============================================================
-- PERSONAGE
-- ============================================================

CREATE TABLE personage (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    first_name TEXT NOT NULL,
    last_name TEXT NOT NULL,

    title person_title,

    language_region_code TEXT
        REFERENCES language_region(code),

    phone_number TEXT,
    email_address TEXT,

    photo_path TEXT
);


CREATE INDEX idx_personage_email
    ON personage(email_address);

CREATE INDEX idx_personage_name
    ON personage(last_name, first_name);


CREATE TABLE personage_link (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    personage_id BIGINT NOT NULL
        REFERENCES personage(id)
        ON DELETE CASCADE,

    url TEXT NOT NULL,
    sort_order INTEGER NOT NULL DEFAULT 0,

    UNIQUE (personage_id, url)
);


-- ============================================================
-- VIEWER : PERSONAGE
-- ============================================================

CREATE TABLE viewer (
    personage_id BIGINT PRIMARY KEY
        REFERENCES personage(id)
        ON DELETE CASCADE
);


-- ============================================================
-- MEMBER : PERSONAGE
-- ============================================================

CREATE TABLE member (
    personage_id BIGINT PRIMARY KEY
        REFERENCES personage(id)
        ON DELETE CASCADE
);


-- ============================================================
-- MANAGER : MEMBER
-- ============================================================

CREATE TABLE manager (
    personage_id BIGINT PRIMARY KEY
        REFERENCES member(personage_id)
        ON DELETE CASCADE
);


-- ============================================================
-- ORGANIZATION POSITION
--
-- Each Organization has its own position list.
-- ============================================================

CREATE TABLE organization_position (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    organization_id BIGINT NOT NULL
        REFERENCES organization(company_id)
        ON DELETE CASCADE,

    name TEXT NOT NULL,

    UNIQUE (organization_id, name),
    UNIQUE (id, organization_id)
);


-- ============================================================
-- ORGANIZATION WORKER
--
-- Personage::p_Company
-- Personage::p_Position
-- Personage::p_JobStatus
--
-- are represented by this relationship.
-- ============================================================

CREATE TABLE organization_worker (
    organization_id BIGINT NOT NULL,

    personage_id BIGINT NOT NULL,

    position_id BIGINT,

    job_status job_status NOT NULL,

    PRIMARY KEY (
        organization_id,
        personage_id
    ),

    FOREIGN KEY (organization_id)
        REFERENCES organization(company_id)
        ON DELETE CASCADE,

    FOREIGN KEY (personage_id)
        REFERENCES personage(id)
        ON DELETE CASCADE,

    FOREIGN KEY (
        position_id,
        organization_id
    )
        REFERENCES organization_position(
            id,
            organization_id
        )
);

-- Your C++ structure currently has one Company per Personage.
-- This enforces one Organization per worker.
CREATE UNIQUE INDEX uq_organization_worker_person
    ON organization_worker(personage_id);

-- ============================================================
-- MEMBER AUTHORITIES
-- ============================================================

CREATE TABLE member_authority (
    member_id BIGINT NOT NULL
        REFERENCES member(personage_id)
        ON DELETE CASCADE,

    authority authority NOT NULL,

    PRIMARY KEY (
        member_id,
        authority
    )
);

-- ============================================================
-- MEMBER <-> CLIENT GROUP
--
-- Member::p_Clients
-- ============================================================

CREATE TABLE member_client_group (
    member_id BIGINT NOT NULL
        REFERENCES member(personage_id)
        ON DELETE CASCADE,

    client_group_id BIGINT NOT NULL
        REFERENCES client_group(company_id)
        ON DELETE CASCADE,

    PRIMARY KEY (
        member_id,
        client_group_id
    )
);

-- ============================================================
-- MANAGER <-> MEMBER
--
-- Manager::p_Team
-- ============================================================

CREATE TABLE manager_team (
    manager_id BIGINT NOT NULL
        REFERENCES manager(personage_id)
        ON DELETE CASCADE,

    member_id BIGINT NOT NULL
        REFERENCES member(personage_id)
        ON DELETE CASCADE,

    PRIMARY KEY (
        manager_id,
        member_id
    ),

    CHECK (manager_id <> member_id)
);

-- ============================================================
-- CLIENT GROUP <-> VIEWER
--
-- ClientGroup::p_Viewers
-- ============================================================

CREATE TABLE client_group_viewer (
    client_group_id BIGINT NOT NULL
        REFERENCES client_group(company_id)
        ON DELETE CASCADE,

    viewer_id BIGINT NOT NULL
        REFERENCES viewer(personage_id)
        ON DELETE CASCADE,

    PRIMARY KEY (
        client_group_id,
        viewer_id
    )
);

-- ============================================================
-- ISSUE
--
-- IMPORTANT:
--
-- Issue is independent.
--
-- It may refer to:
--
--   Organization only
--   ClientGroup only
--   Organization + ClientGroup
--
-- It does NOT belong to ClientGroup.
-- ============================================================

CREATE TABLE issue (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    organization_id BIGINT
        REFERENCES organization(company_id)
        ON DELETE RESTRICT,

    client_group_id BIGINT
        REFERENCES client_group(company_id)
        ON DELETE RESTRICT,

    member_id BIGINT
        REFERENCES member(personage_id)
        ON DELETE SET NULL,

    viewer_id BIGINT
        REFERENCES viewer(personage_id)
        ON DELETE SET NULL,

    status issue_status NOT NULL,

    start_date TIMESTAMPTZ NOT NULL,

    end_date TIMESTAMPTZ,

    description JSONB,

    parent_issue_id BIGINT
        REFERENCES issue(id)
        ON DELETE SET NULL,

    CHECK (
        organization_id IS NOT NULL
        OR client_group_id IS NOT NULL
    ),

    CHECK (
        end_date IS NULL
        OR end_date >= start_date
    )
);


CREATE INDEX idx_issue_organization
    ON issue(organization_id);

CREATE INDEX idx_issue_client_group
    ON issue(client_group_id);

CREATE INDEX idx_issue_member
    ON issue(member_id);

CREATE INDEX idx_issue_viewer
    ON issue(viewer_id);

CREATE INDEX idx_issue_status
    ON issue(status);

CREATE INDEX idx_issue_parent
    ON issue(parent_issue_id);


-- ============================================================
-- PROFIT
--
-- Independent from both Organization and ClientGroup.
--
-- At least one reference is required.
-- ============================================================

CREATE TABLE profit (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    organization_id BIGINT
        REFERENCES organization(company_id)
        ON DELETE RESTRICT,

    client_group_id BIGINT
        REFERENCES client_group(company_id)
        ON DELETE RESTRICT,

    amount NUMERIC(19, 4) NOT NULL,

    currency_code TEXT NOT NULL
        REFERENCES currency(code),

    issue_date TIMESTAMPTZ NOT NULL,

    description JSONB,

    CHECK (
        organization_id IS NOT NULL
        OR client_group_id IS NOT NULL
    )
);


CREATE INDEX idx_profit_organization_date
    ON profit(organization_id, issue_date);

CREATE INDEX idx_profit_client_group_date
    ON profit(client_group_id, issue_date);


-- ============================================================
-- CHURN
--
-- Independent from both Organization and ClientGroup.
--
-- At least one reference is required.
-- ============================================================

CREATE TABLE churn (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    organization_id BIGINT
        REFERENCES organization(company_id)
        ON DELETE RESTRICT,

    client_group_id BIGINT
        REFERENCES client_group(company_id)
        ON DELETE RESTRICT,

    issue_date TIMESTAMPTZ NOT NULL,

    description JSONB,

    CHECK (
        organization_id IS NOT NULL
        OR client_group_id IS NOT NULL
    )

CREATE INDEX idx_churn_organization_date
    ON churn(organization_id, issue_date);

CREATE INDEX idx_churn_client_group_date
    ON churn(client_group_id, issue_date);

-- ============================================================
-- SUBSCRIPTION
-- ============================================================

CREATE TABLE subscription (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    organization_id BIGINT NOT NULL
        REFERENCES organization(company_id)
        ON DELETE RESTRICT,

    issue_date TIMESTAMPTZ NOT NULL,

    type subscription_type NOT NULL
);

CREATE INDEX idx_subscription_organization_date
    ON subscription(
        organization_id,
        issue_date
    );

-- ============================================================
-- VALIDATION
--
-- If Organization and ClientGroup are BOTH specified in an
-- Issue / Profit / Churn, make sure the ClientGroup actually
-- belongs to that Organization.
-- ============================================================

CREATE OR REPLACE FUNCTION validate_organization_client_group()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
    actual_organization_id BIGINT;
BEGIN

    IF NEW.client_group_id IS NULL THEN
        RETURN NEW;
    END IF;

    IF NEW.organization_id IS NULL THEN
        RETURN NEW;
    END IF;

    SELECT organization_id
    INTO actual_organization_id
    FROM client_group
    WHERE company_id = NEW.client_group_id;

    IF actual_organization_id IS DISTINCT FROM NEW.organization_id THEN
        RAISE EXCEPTION
            'ClientGroup % does not belong to Organization %',
            NEW.client_group_id,
            NEW.organization_id;
    END IF;

    RETURN NEW;
END;
$$;


CREATE TRIGGER trg_issue_validate_scope
BEFORE INSERT OR UPDATE
ON issue
FOR EACH ROW
EXECUTE FUNCTION validate_organization_client_group();


CREATE TRIGGER trg_profit_validate_scope
BEFORE INSERT OR UPDATE
ON profit
FOR EACH ROW
EXECUTE FUNCTION validate_organization_client_group();


CREATE TRIGGER trg_churn_validate_scope
BEFORE INSERT OR UPDATE
ON churn
FOR EACH ROW
EXECUTE FUNCTION validate_organization_client_group();
